def coordinate(row):
 if isinstance(row,tuple):return row
 return row['chromosome'],int(row['start']),int(row['end']),row['gene']
def valid_keys(rows,wanted):
 keys=[coordinate(row) for row in rows]
 assert keys and len(keys)==len(set(keys)),'empty/duplicate bin coordinates'
 assert Counter(keys)==Counter(wanted),'missing/extra/incorrect bin coordinates'
 assert all(c=='chrAudit' and 0<=a<b<=1000000 for c,a,b,g in keys),'invalid coordinate system/bounds'
 return keys

def geometry(targets,antitargets):
 for c,a,b,g in targets:
  for cc,aa,bb,gg in antitargets:
   assert c!=cc or min(b,bb)<=max(a,aa),'target-antitarget intersection'

def coverage(rows,wanted,sample):
 keys=valid_keys(rows,wanted)
 for row,(_,start,end,_) in zip(rows,keys):
  # Count aligned-base overlap ourselves; CNVkit's coverage routine is never called here.
  expected_depth=Fraction(sum(max(0,min(end,b)-max(start,a)) for a,b in inputs[sample]),end-start)
  assert expected_depth==truth['samples'][sample]['uniform_expected_depth']
  assert Fraction(Decimal(row['depth']))==expected_depth,'incorrect aligned-base depth'
  assert Decimal(row['log2'])==(0 if expected_depth==1 else 1),'incorrect raw log2 depth'
  assert all(Decimal(row[k]).is_finite() for k in ['depth','log2'])
 return keys

def bed(path):
 rows=[]
 for line in path.read_text().splitlines():
  fields=line.split('\t');assert len(fields)==4
  rows.append((fields[0],int(fields[1]),int(fields[2]),fields[3]))
 return rows

def cnn(path):
 with path.open() as f:rows=list(csv.DictReader(f,delimiter='\t'))
 assert rows,'empty coverage/reference table'
 return rows
